document.getElementById('upload').addEventListener('change', e => {
  const file = e.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = evt => {
    const img = new Image();
    img.src = evt.target.result;
    img.onload = () => {
      const tex = new THREE.Texture(img);
      tex.needsUpdate = true;
      sticker.material.map = tex;
      sticker.material.needsUpdate = true;
    };
  };
  reader.readAsDataURL(file);
});

// Drag and drop
const raycaster = new THREE.Raycaster();
const mouse = new THREE.Vector2();
let dragging = false;

window.addEventListener('mousedown', e => {
  mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
  mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;
  raycaster.setFromCamera(mouse, camera);
  const hits = raycaster.intersectObjects([sticker]);
  if (hits.length) dragging = true;
});

window.addEventListener('mousemove', e => {
  if (!dragging) return;
  mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
  mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;
  raycaster.setFromCamera(mouse, camera);
  const plane = new THREE.Plane(new THREE.Vector3(0, 1, 0), 0);
  const point = raycaster.ray.intersectPlane(plane);
  if (point) sticker.position.set(point.x, 0.5, point.z);
});

window.addEventListener('mouseup', () => dragging = false);
